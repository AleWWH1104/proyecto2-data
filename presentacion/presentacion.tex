\begin{frame}{Problema, objetivo y resultado principal}
  \begin{block}{Segmentación, no clasificación de órganos}
    Dada una imagen histológica, producir una máscara binaria de unidades
    funcionales de tejido (FTU): FTU frente a fondo. El órgano es metadato,
    no una clase predicha. Contexto: Jain et al. (2023) \cite{jain}.
  \end{block}
  \begin{itemize}
    \item Comparar tres arquitecturas con el mismo preprocesamiento y split.
    \item Medir solapamiento, diferencias por órgano y tiempo registrado.
    \item Integrar exploración, resultados e ingreso de imágenes en Streamlit.
  \end{itemize}
  \alert{SegFormer / MIT-B0 obtiene el mayor Dice medio registrado: 0.629.}
  La predicción local con pesos reales sigue pendiente de recuperar checkpoints.
\end{frame}

\begin{frame}{Entrada compartida: de la histología al tensor}
  \begin{enumerate}
    \item Conversión RGB uint8; referencia física de $0.4\,\mu$m/píxel.
    \item Redimensionamiento a $384\times384$; máscaras con vecino más cercano.
    \item Normalización ImageNet: media $(0.485,0.456,0.406)$ y
          desviación $(0.229,0.224,0.225)$.
    \item Solo entrenamiento: volteos, rotación, escala y cambios de color,
          brillo, gamma y desenfoque. Validación/prueba: sin aumentación aleatoria.
  \end{enumerate}
  \begin{block}{Qué recibe el modelo}
    Píxeles de la imagen; edad, sexo, órgano y cobertura RLE son descriptivos.
    La máscara es el objetivo, nunca una entrada de inferencia.
  \end{block}
  \small Aumentación de tinción como motivación: Tellez et al. (2019) \cite{tellez}.
  No se implementa una normalización de tinción dedicada.
\end{frame}

\begin{frame}{Tres modelos: comparación controlada}
  \begin{description}
    \item[U-Net / ResNet34] Línea base encoder--decoder con conexiones de salto:
      detalle local y contexto \cite{unet}.
    \item[U-Net++ / ResNet34] Conexiones de salto rediseñadas para explotar
      características multiescala \cite{unetpp}.
    \item[SegFormer / MIT-B0] Encoder transformer jerárquico y decoder ligero;
      alternativa compacta a las redes convolucionales \cite{segformer}.
  \end{description}
  \begin{block}{Condiciones comunes}
    Encoder preentrenado; salida de un canal, logits binarios;
    misma resolución, split, pérdida y umbral.
    Es una comparación de estas ejecuciones, no de todo el potencial de cada familia.
  \end{block}
\end{frame}

\begin{frame}{Cómo leer las métricas y el tiempo}
  \[
    \mathrm{Dice}=\frac{2|P\cap G|}{|P|+|G|},\qquad
    \mathrm{IoU}=\frac{|P\cap G|}{|P\cup G|}
  \]
  \begin{itemize}
    \item $P$: máscara predicha; $G$: anotación. Rango $[0,1]$, mayor es mejor.
    \item Sigmoid y umbral $\geq0.5$; ambas máscaras vacías reciben 1.
    \item Media por imagen: 53 imágenes comunes, 159 filas para tres modelos.
          No es Dice calculado sobre todos los píxeles agregados.
    \item Tiempo del forward registrado en RTX 4060 Laptop, batch 1:
          excluye carga, preprocesamiento y presentación de resultados.
  \end{itemize}
  \alert{No es un benchmark del Mac ni una medida de latencia extremo a extremo.}
\end{frame}

\begin{frame}{Selección: SegFormer como opción de referencia}
  \begin{block}{Evidencia de esta ejecución}
    Mayor mejor Dice de validación (0.6596), mayor Dice e IoU medios de prueba
    (0.629 y 0.510) y mayor Dice medio en cada órgano registrado.
  \end{block}
  \begin{itemize}
    \item El checkpoint se eligió con validación, no con prueba.
    \item Mantener U-Net y U-Net++ como comparadores en la aplicación.
    \item Forward medio parecido a U-Net; no declarar superioridad de velocidad
          con una diferencia de solo 0.2 ms y sin estudio de variabilidad.
    \item La recomendación es local a este split y perfil; no valida desempeño
          clínico ni generalización a otros laboratorios.
  \end{itemize}
\end{frame}

\begin{frame}{Nueva imagen: flujo disponible al recuperar los pesos}
  \begin{enumerate}
    \item PNG/JPEG/TIFF de una sola imagen; máximo 64 MiB y 9 millones de píxeles.
    \item Preprocesamiento compartido a 384 px; si falta espaciado,
          asumir explícitamente $0.4\,\mu$m/píxel, no inferirlo.
    \item Inferencia CPU, uno o todos los modelos disponibles, secuencialmente.
    \item Máscara binaria restaurada al tamaño original, superposición cian,
          porcentaje FTU y tiempo de forward CPU.
  \end{enumerate}
  \begin{alertblock}{Estado local y alcance}
    Sin los tres archivos \texttt{.pt}, no hay nueva predicción real demostrada.
    Pruebas sintéticas no sustituyen aceptación con pesos reales.
    Sin máscara de referencia no se muestra Dice/IoU de la carga nueva.
  \end{alertblock}
\end{frame}

\begin{frame}{Límites y pendientes de entrega}
  \begin{itemize}
    \item Muestra pequeña: 53 imágenes de prueba, solo 7 de pulmón;
          una ejecución por modelo y sin incertidumbre cuantificada.
    \item Datos HPA y split por imagen: no validar transferencia de dominio
          a HuBMAP, independencia por donante ni uso clínico.
    \item Resolución 384: compromete detalle fino; no se estudiaron otras resoluciones.
    \item Faltan localmente \texttt{train.csv}, caché y checkpoints;
          recuperación y demostración con pesos reales pendientes.
    \item Informe final y capturas reales completos; enlace de Drive,
          permisos y entrega humana externa pendientes de confirmar.
  \end{itemize}
  \alert{Prototipo de investigación: no usar para diagnóstico ni decisiones médicas.}
\end{frame}
